Executive Summary

Beyond the biggest content hits, Netflix viewing performance extends with older titles maintaining steady viewing across reporting periods (2023 - 2026 first half). The big hits and its series also brought Netflix steady growing views. For 2027 and beyond, Netflix should prioritize original series with sustainable viewing potential, and keeping a mix of movies and regional contents for its diverse audiences.

Task 1: Data Acquisition

From an advertising industry perspective and a creator profit-participation perspective, analyzing the popularity of Netflix content and understanding the insights behind its viewing figures are essential for several purposes, including investment allocation, fair compensation for talent and production crews, and maintaining healthy competition. Netflix releases its What We Watched: A Netflix Engagement Report biannually to enhance industry transparency. The report publishes global streaming data covering 99% of all viewing on the platform, including titles watched hours.

Code 
load_netflix_data <- function(){
 
  library(tidyverse)
  library(rvest)
  library(readxl)
  library(fs)
  
  data_dir <- path("data", "mp01")
    
  if(!dir_exists(data_dir)){
    dir_create(data_dir, showWarnings=FALSE, recurse=TRUE)
  }
  
  NETFLIX_URLS <- list(
    `2023H1` = "https://about.netflix.com/news/what-we-watched-a-netflix-engagement-report",
    `2023H2` = "https://about.netflix.com/news/what-we-watched-the-second-half-of-2023",
    `2024H1` = "https://about.netflix.com/news/what-we-watched-the-first-half-of-2024",
    `2024H2` = "https://about.netflix.com/news/what-we-watched-the-second-half-of-2024",
    `2025H1` = "https://about.netflix.com/news/what-we-watched-the-first-half-of-2025",
    `2025H2` = "https://about.netflix.com/news/what-we-watched-the-second-half-of-2025",
    `2026H1` = "https://about.netflix.com/news/what-we-watched-the-first-half-of-2026"
  )
  
  download_if_needed <- function(url, local){
    if(!file_exists(local)){
      download.file(url, local, quiet = TRUE, mode = "wb")
    }
    
    return(local)
  }
    
NETFLIX_URLS |> 
    map(read_html) |> 
    map(html_element, xpath = "//a[contains(@href, '.xlsx')]") |> 
    map_chr(html_attr, "href") |> 
    enframe("period", "url") |> 
    mutate(path = path(data_dir, paste0(period, ".xlsx")), 
           path = map2(url, path, download_if_needed), 
           sheets = map(path, excel_sheets)) |>
    unnest(sheets) |> 
    mutate(data = map2(path, sheets, \(x, y) read_excel(x, sheet=y, skip=3, na="*")), 
           data = map(data, \(x) x |> mutate(`Release Date` = as.Date(`Release Date`))), 
           data = map2(data, sheets, \(x, y) x |> mutate(sheet=y))) |> 
    unnest(data) |> 
    select(-url, -path, -sheets) |> 
    mutate(year=str_sub(period, 1, 4), 
           half=str_sub(period, start=6), 
           period_start = make_date(year, if_else(half==1, 1, 7), 1), 
           period_end = make_date(year, if_else(half==1, 6, 12), if_else(half==1, 30, 31))) |> 
    filter(!is.na(`Hours Viewed`)) |>
    mutate(live_event = !is.na(Runtime) & str_detect(Runtime, "\\d+:\\d+\\*"), 
           clean_time = str_remove(Runtime, fixed("*")),
           Runtime = 60 * hour(hm(clean_time)) + minute(hm(clean_time)),
           title_type = replace_values(sheet, 
                                      "Shows" ~ "TV",
                                      "Film" ~ "Movies",
                                      "Engagement" ~ NA)) |>
    select(-clean_time, -sheet) |>
    mutate(Runtime = coalesce(Runtime, mean(Runtime, na.rm=TRUE)), 
           .by=Title, 
           Title = stringi::stri_trans_general(Title, "Latin-ASCII"),
           Title = str_remove(Title, " // .*"), 
           Title = str_remove(Title, "[^[:ascii:]].*"), 
           Title = str_trim(Title)) |>
    filter_out(Title == "") |>
    select(year, 
           period, 
           period_start, 
           period_end, 
           Title, 
           -half, 
           Runtime, 
           Views, 
           `Hours Viewed`, 
           `Release Date`,
           `Available Globally?`, 
           live_event, 
           title_type) |> 
    group_by(Title) |> 
    fill(title_type,
         .direction="updown") |>
    ungroup()
}

NETFLIX_DATA <- load_netflix_data()

6.3 Consistently Popular Titles

This analysis looks at which titles keep strong and steady viewing across all seven reporting periods. I include titles that appear in every period, have positive average viewing hours, and have a coefficient of variation (CV) of 50% or less. CV compares the standard deviation with the average viewing hours, so a lower value means viewing is more stable across periods. I then rank the top 10 by their minimum viewing hours, highlighting titles that maintain strong viewing even in their lowest-performing period.

Code 
number_of_periods <- n_distinct(NETFLIX_DATA$period)

consistent_titles <- NETFLIX_DATA |>
  group_by(title, period) |>
  summarize(hours = sum(hours_viewed, na.rm = TRUE), .groups = "drop") |>
  group_by(title) |>
  summarize(periods = n(),
            average_hours = mean(hours),
            minimum_hours = min(hours),
            sd_hours = sd(hours),
            .groups = "drop") |>
  mutate(cv = sd_hours / average_hours) |>
  filter(periods == number_of_periods, average_hours > 0, cv <= 0.5) |>
  slice_max(minimum_hours, n = 10, with_ties = FALSE) |>
  arrange(desc(minimum_hours))

consistent_titles |>
  select(title, average_hours, minimum_hours, sd_hours, cv) |>
  gt() |>
  tab_header(title = "Consistently Popular Titles") |>

6.4 Shows vs. Movies

The dataset does not include a flag to distinguish show or movie, so I used title names and runtime to estimate content type. Titles containing “Season” or having a runtime longer than 240 minutes are classified as shows, since these features may indicate content with multiple episodes. Other titles with a known runtime are classified as movies. Titles without “Season” in their names and with missing runtimes are classified as Unknown. I then compare each category’s share of total viewing hours across reporting periods.

Code 
content_shares <- NETFLIX_DATA |>
  mutate(content_type = case_when(
    str_detect(title, regex("Season", ignore_case = TRUE)) ~ "Show",
    runtime > 240 ~ "Show",
    !is.na(runtime) ~ "Movie",
    TRUE ~ "Unknown"
  )) |>
  group_by(period, content_type) |>
  summarize(hours = sum(hours_viewed, na.rm = TRUE), .groups = "drop") |>
  group_by(period) |>
  mutate(share = hours / sum(hours)) |>
  ungroup() |>
  select(period, content_type, share) |>
  pivot_wider(names_from = content_type, values_from = share, values_fill = 0)

  gt() |>

  tab_source_note(paste(

6.5 Live vs. Pre-Taped Content

The analysis below compares live and pre-taped content using average viewing hours and the 90th percentile within each period and content type. The average describes overall viewing hours per entry, while the 90th percentile represents the threshold for approximately the highest-performing 10% of entries. I use this threshold to compare strong performance without relying only on the single most-watched title.

Code 
live_comparison <- NETFLIX_DATA |>
  mutate(content_type = if_else(live_event, "Live", "Pre-taped")) |>
  group_by(period, content_type) |>
  summarize(
    entries = n(),
    average_hours = mean(hours_viewed, na.rm = TRUE),
    top_tier_hours = quantile(hours_viewed, probs = 0.9, na.rm = TRUE, names = FALSE),
    .groups = "drop"
  )

live_comparison |>
  gt(groupname_col = "period") |>
  tab_header(title = "Live Events Compared with Pre-Taped Content") |>
  cols_label(content_type = "Content Type", entries = "Title Entries",

```{r load-data}
load_netflix_data <- function(){
 
  library(tidyverse)
  library(rvest)
  library(readxl)
  library(fs)
  
  data_dir <- path("data", "mp01")
    
  if(!dir_exists(data_dir)){
    dir_create(data_dir, showWarnings=FALSE, recurse=TRUE)
  }
  
  NETFLIX_URLS <- list(
    `2023H1` = "https://about.netflix.com/news/what-we-watched-a-netflix-engagement-report",
    `2023H2` = "https://about.netflix.com/news/what-we-watched-the-second-half-of-2023",
    `2024H1` = "https://about.netflix.com/news/what-we-watched-the-first-half-of-2024",
    `2024H2` = "https://about.netflix.com/news/what-we-watched-the-second-half-of-2024",
    `2025H1` = "https://about.netflix.com/news/what-we-watched-the-first-half-of-2025",
    `2025H2` = "https://about.netflix.com/news/what-we-watched-the-second-half-of-2025",
    `2026H1` = "https://about.netflix.com/news/what-we-watched-the-first-half-of-2026"
  )
  
  download_if_needed <- function(url, local){
    if(!file_exists(local)){
      download.file(url, local, quiet = TRUE, mode = "wb")
    }
    
    return(local)
  }
    
NETFLIX_URLS |> 
    map(read_html) |> 
    map(html_element, xpath = "//a[contains(@href, '.xlsx')]") |> 
    map_chr(html_attr, "href") |> 
    enframe("period", "url") |> 
    mutate(path = path(data_dir, paste0(period, ".xlsx")), 
           path = map2(url, path, download_if_needed), 
           sheets = map(path, excel_sheets)) |>
    unnest(sheets) |> 
    mutate(data = map2(path, sheets, \(x, y) read_excel(x, sheet=y, skip=3, na="*")), 
           data = map(data, \(x) x |> mutate(`Release Date` = as.Date(`Release Date`))), 
           data = map2(data, sheets, \(x, y) x |> mutate(sheet=y))) |> 
    unnest(data) |> 
    select(-url, -path, -sheets) |> 
    mutate(year=str_sub(period, 1, 4), 
           half=str_sub(period, start=6), 
           period_start = make_date(year, if_else(half==1, 1, 7), 1), 
           period_end = make_date(year, if_else(half==1, 6, 12), if_else(half==1, 30, 31))) |> 
    filter(!is.na(`Hours Viewed`)) |>
    mutate(live_event = !is.na(Runtime) & str_detect(Runtime, "\\d+:\\d+\\*"), 
           clean_time = str_remove(Runtime, fixed("*")),
           Runtime = 60 * hour(hm(clean_time)) + minute(hm(clean_time)),
           title_type = replace_values(sheet, 
                                      "Shows" ~ "TV",
                                      "Film" ~ "Movies",
                                      "Engagement" ~ NA)) |>
    select(-clean_time, -sheet) |>
    mutate(Runtime = coalesce(Runtime, mean(Runtime, na.rm=TRUE)), 
           .by=Title, 
           Title = stringi::stri_trans_general(Title, "Latin-ASCII"),
           Title = str_remove(Title, " // .*"), 
           Title = str_remove(Title, "[^[:ascii:]].*"), 
           Title = str_trim(Title)) |>
    filter_out(Title == "") |>
    select(year, 
           period, 
           period_start, 
           period_end, 
           Title, 
           -half, 
           Runtime, 
           Views, 
           `Hours Viewed`, 
           `Release Date`,
           `Available Globally?`, 
           live_event, 
           title_type) |> 
    group_by(Title) |> 
    fill(title_type,
         .direction="updown") |>
    ungroup()
}
```
